\chapter{نمونه جریان اجرای سامانه}

\section{جریان یک نوبت مکالمه}

جریان ساده‌شده اجرای یک نوبت مکالمه به‌صورت زیر است:

\begin{latin}\begin{verbatim}
1. Receive user audio
2. Convert to 16 kHz mono PCM16
3. Apply gated low-level gain when needed
4. Run Vosk primary and N-best decoding
5. Normalize all Persian hypotheses
6. Apply contextual correction memory
7. Assess confidence and critical-slot risk
8. Retry, clarify, or construct the LLM context
9. Run Gemma 2 9B when an answer is permitted
10. Update the bounded session history
11. Run Piper TTS and play the response
12. Record confidence, decisions, latency, and errors
\end{verbatim}\end{latin}


\section{نمونه شبه‌کد}

\begin{latin}\begin{verbatim}
audio = get_audio()

raw = transcribe_audio_detailed(audio)
transcription = preprocess_transcription(raw)
decision = interpret_transcription(transcription, session)

if decision.action == "retry":
    response = retry_message
elif decision.action == "clarify":
    response = decision.risk.clarification
else:
    response = llm.generate(
        decision.effective_question,
        history=session.messages(),
        memory=session.memory_prompt(),
    )
    session.add_turn(decision.effective_question, response)

speech = tts.synthesize(response)

play_audio(speech)
\end{verbatim}\end{latin}
